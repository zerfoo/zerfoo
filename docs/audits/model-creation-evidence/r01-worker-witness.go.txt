//go:build r01audit

package cli

import (
	"math"
	"os"
	"path/filepath"
	"testing"

	"github.com/zerfoo/zerfoo/training/automl"
)

// This audit witness characterizes a defect, not desired behavior. Keep it out
// of ordinary CI; replace with the isolated-evaluation regression in R05.
func TestR01ProductionWorkerEvaluationLeakage(t *testing.T) {
	path := filepath.Join(t.TempDir(), "balanced.csv")
	if err := os.WriteFile(path, []byte("x,label\n0,0\n0,0\n0,1\n0,1\n0,2\n0,2\n"), 0600); err != nil {
		t.Fatal(err)
	}
	for _, metric := range []string{"accuracy", "loss", "unsupported"} {
		t.Run(metric, func(t *testing.T) {
			worker, err := newTabularWorker(path, metric)
			if err != nil {
				t.Fatal(err)
			}
			got, err := worker.RunTrial(automl.Config{Params: map[string]float64{"lr": 0.001, "batch_size": 5}})
			if err != nil {
				t.Fatal(err)
			}
			// Every input is identical, hence every prediction has the same class.
			// Full-data accuracy is exactly 2/6. Train's 20% validation split has
			// one row, whose accuracy can only be 0 or 1, never 1/3.
			if math.Abs(got.Score-1.0/3) > 1e-12 {
				t.Fatalf("baseline changed: score=%g; investigate and replace audit witness", got.Score)
			}
			t.Logf("CONFIRMED baseline defect: requested=%s score=%g; one-row held-out accuracy must be 0 or 1", metric, got.Score)
		})
	}
}
